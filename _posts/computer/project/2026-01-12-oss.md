

阿里云使用oss的临时凭证问题

https://help.aliyun.com/zh/oss/developer-reference/use-temporary-access-credentials-provided-by-sts-to-access-oss#2c550502c3j2n





1. 假设只提供以下内容

   ```
     string policy = 1;//长度约1356字符串
     string security_token = 2; //长度约620
     string signature_version = 3; //加密方法
     string credential = 4;//STS.wangfuyu/20251018/cn-shanghai/oss/aliyun_v4_request
     string date = 5;  //20251018T123944Z
     string signature = 6; //长度64位字符串
     string host = 7; //wangfuyu.oss-cn-shanghai.aliyuncs.com
   ```

   服务端使用的是：

   ```
   import ("github.com/aliyun/aliyun-oss-go-sdk/oss"）
   ..
   aliOss, _ := oss.New(
   		stsConf.Endpoint,//oss-cn-shanghai.aliyuncs.com
   		stsConf.AccessKeyId, //24个长度
   		stsConf.AccessKeySecret, //大概是30个长度
   		oss.SecurityToken(stsConf.SecurityToken))
   	aliBucket, _ := aliOss.Bucket(stsConf.Bucket)
   	
   ```
